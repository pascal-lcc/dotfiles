# split windows like vim

bind -r K resize-pane -U 2
bind -r L resize-pane -R 2

# 1. 基礎身分定義 (避免噴出 WezTerm 查詢亂碼)

# 2. 解決 nvim 的 Esc 鍵延遲 (非常重要！)

# 2. 修正 overrides。
